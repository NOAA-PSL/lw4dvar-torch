# Rebuild (login node):
#   1. conda create -p <env> --file aifs3-spec.txt
#      NOTE: the rdhpcs-conda module's system .condarc pins pkgs_dirs to
#      ~/.conda/pkgs with #!final (CONDA_PKGS_DIRS is ignored) and this env
#      downloads ~14 GB -- use a conda without that config, e.g.
#      CONDA_PKGS_DIRS=<scratch dir> /scratch4/BMC/gsienkf/Sergey.Frolov/eagle/latent/EAGLE/conda/bin/conda
#   2. <env>/bin/python -m pip install --no-deps --no-cache-dir -r aifs3-requirements.txt
#   3. anemoi from NOAA-PSL forks at the commits used for training, with the
#      versions recorded in the checkpoint (setuptools_scm can't derive them
#      from a clone of the monorepo):
#        git clone https://github.com/NOAA-PSL/anemoi-core.git      # @ 7b53fcb5f
#        git clone https://github.com/NOAA-PSL/anemoi-inference.git # @ 2169c56
#        SETUPTOOLS_SCM_PRETEND_VERSION_FOR_ANEMOI_MODELS=0.19.0.post35     pip install --no-deps anemoi-core/models
#        SETUPTOOLS_SCM_PRETEND_VERSION_FOR_ANEMOI_GRAPHS=0.9.7.post35      pip install --no-deps anemoi-core/graphs
#        SETUPTOOLS_SCM_PRETEND_VERSION_FOR_ANEMOI_TRAINING=0.17.0.post35   pip install --no-deps anemoi-core/training
